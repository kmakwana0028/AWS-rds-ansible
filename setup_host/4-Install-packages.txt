## Install Jenkins:
    ## Configure the Security Group: Ensure that port 8080 (Jenkins' default port) is open
       in the security group associated with your EC2 instance to allow inbound traffic.

    sudo dnf install java-11-amazon-corretto-devel -y #11 Or a higher compatible java version
    Or
    sudo dnf install java-17-amazon-corretto -y    
    java -version # Verify the java version


    # Download Jenkins files

    sudo wget -O /etc/yum.repos.d/jenkins.repo https://pkg.jenkins.io/redhat-stable/jenkins.repo
    # Import the Jenkins GPG key
    sudo rpm --import https://pkg.jenkins.io/redhat-stable/jenkins.io-2023.key


    sudo dnf install jenkins -y # Install Jenkins
        sudo systemctl enable jenkins
        sudo systemctl start jenkins
        sudo systemctl status jenkins
    # Complete Post-Installation Setup:
        # Open a web browser and navigate to your EC2 instance's public IP address or domain name, followed by port 8080 (e.g., http://your-ec2-ip:8080).
        # Retrieve the initial administrator password from the Jenkins log file below
        # Access : http://your_amazon_linux_instance_ip:8080

    sudo cat /var/lib/jenkins/secrets/initialAdminPassword
